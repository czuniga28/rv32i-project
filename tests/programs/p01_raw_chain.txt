# p01 - Hazard RAW a distancia 1: cada instrucción usa el resultado de la anterior
# Esperado: x1 = 8, x2 = 9, x3 = 1, x4 = 2, x5 = 32, x6 = 8
addi  x1, x0, 1          # 0x00  x1 = 1
add   x1, x1, x1         # 0x04  2  (rs1 y rs2 dependen, distancia 1)
add   x1, x1, x1         # 0x08  4
add   x1, x1, x1         # 0x0c  8
addi  x2, x1, 1          # 0x10  9
sub   x3, x2, x1         # 0x14  1  (rs1 a distancia 1, rs2 a distancia 2)
add   x4, x3, x3         # 0x18  2
slli  x5, x4, 4          # 0x1c  32
srl   x6, x5, x4         # 0x20  32 >> 2 = 8 (rs1 a distancia 1, rs2 a distancia 2)
ebreak                   # 0x24
