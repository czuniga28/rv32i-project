# t03 - ALU suma/resta: desbordamientos y acarreos se descartan (mod 2^32)
lui   x1, 0x80000        # 0x00  INT_MIN = 0x80000000
addi  x2, x1, -1         # 0x04  INT_MAX = 0x7fffffff (resta que da la vuelta)
addi  x3, x2, 1          # 0x08  INT_MAX + 1 = 0x80000000
add   x4, x2, x2         # 0x0c  0xfffffffe
sub   x5, x1, x2         # 0x10  INT_MIN - INT_MAX = 1
sub   x6, x0, x1         # 0x14  -INT_MIN = 0x80000000
sub   x7, x0, x2         # 0x18  -INT_MAX = 0x80000001
addi  x8, x0, -1         # 0x1c  0xffffffff
add   x9, x8, x8         # 0x20  0xfffffffe
addi  x10, x8, 1         # 0x24  0 (acarreo descartado)
sub   x11, x2, x8        # 0x28  INT_MAX - (-1) = 0x80000000
add   x12, x1, x1        # 0x2c  0
ebreak                   # 0x30
