# t06 - Comparaciones con y sin signo (slt/sltu/slti/sltiu)
addi  x1, x0, -1         # 0x00
addi  x2, x0, 1          # 0x04
slt   x3, x1, x2         # 0x08  1: -1 < 1 con signo
sltu  x4, x1, x2         # 0x0c  0: 0xffffffff < 1 sin signo es falso
slt   x5, x2, x1         # 0x10  0
sltu  x6, x2, x1         # 0x14  1
slt   x7, x1, x1         # 0x18  0: iguales
slti  x8, x1, 0          # 0x1c  1
sltiu x9, x1, -1         # 0x20  0: imm -1 se extiende a 0xffffffff
sltiu x10, x2, -1        # 0x24  1
sltiu x11, x0, 1         # 0x28  1 (patrón seqz)
sltiu x12, x2, 1         # 0x2c  0
lui   x13, 0x80000       # 0x30  INT_MIN
slt   x14, x13, x0       # 0x34  1
sltu  x15, x13, x0       # 0x38  0
slti  x16, x13, -2048    # 0x3c  1
slt   x17, x0, x0        # 0x40  0
ebreak                   # 0x44
